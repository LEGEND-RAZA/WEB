2) Replace your existing restoreRemoteSession() with the function below. It downloads into a temporary directory, verifies creds.json, and only then replaces the local session directory.

async function restoreRemoteSession() {
  if (!SESSION_ID) {
    // Allows local development with an existing ./session directory.
    console.log('[SESSION] SESSION_ID is not set; using local session files.')
    remoteSessionRestored = true
    return false
  }

  console.log('[SESSION] Downloading session from pairing portal...')
  const url = `${PAIR_SERVER_URL.replace(/\/+$/, '')}/api/session/${encodeURIComponent(SESSION_ID)}`
  const response = await fetch(url, {
    headers: { Accept: 'application/json' },
    signal: AbortSignal.timeout(20000)
  })

  if (!response.ok) {
    throw new Error(`Session API returned HTTP ${response.status}. Check SESSION_ID, PAIR_SERVER_URL, SQL session expiry, and portal logs.`)
  }

  const data = await response.json()
  if (!data || data.success !== true || !data.files || typeof data.files !== 'object') {
    throw new Error('Session API returned an invalid payload.')
  }

  const parent = path.dirname(SESSION_DIR)
  const stagingDir = path.join(parent, `.session-restore-${process.pid}-${Date.now()}`)
  await fs.promises.mkdir(stagingDir, { recursive: true })

  try {
    let restored = 0
    for (const [fileName, encoded] of Object.entries(data.files)) {
      // Baileys auth filenames are simple names; never allow a path from API data.
      if (!/^[A-Za-z0-9._-]+$/.test(fileName) || fileName === '.' || fileName === '..') continue
      if (typeof encoded !== 'string' || !/^(?:[A-Za-z0-9+/]{4})*(?:[A-Za-z0-9+/]{2}==|[A-Za-z0-9+/]{3}=)?$/.test(encoded)) continue
      const bytes = Buffer.from(encoded, 'base64')
      if (!bytes.length) continue
      await fs.promises.writeFile(path.join(stagingDir, fileName), bytes, { mode: 0o600 })
      restored++
    }

    if (!restored || !fs.existsSync(path.join(stagingDir, 'creds.json'))) {
      throw new Error('Downloaded session is incomplete: creds.json is missing.')
    }

    // Replace local auth only after the complete remote session has been staged.
    await fs.promises.rm(SESSION_DIR, { recursive: true, force: true })
    await fs.promises.rename(stagingDir, SESSION_DIR)
    remoteSessionRestored = true
    console.log(`[SESSION] Restored ${restored} auth file(s).`)
    return true
  } catch (error) {
    await fs.promises.rm(stagingDir, { recursive: true, force: true }).catch(() => {})
    throw error
  }
}

The complete bot-index.js in this bundle is preferred over applying this patch alone.
